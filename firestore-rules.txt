    // ---------- Aria Holiday (aria_ collections) ----------
    match /aria_settings/{doc} {
      allow read: if true;
      allow write: if request.auth != null;
    }
    match /aria_avail/{code} {
      // room/date blocks only, no guest details, so guests can check availability
      allow read: if true;
      allow create: if true;
      allow update, delete: if request.auth != null;
    }
    match /aria_bookings/{code} {
      allow create: if request.resource.data.code == code
                    && request.resource.data.status == 'pending'
                    && request.resource.data.paidAmount == 0;
      allow get: if true;                       // guest looks up by secret code
      allow list, delete: if request.auth != null;
      allow update: if request.auth != null
        || request.resource.data.diff(resource.data).affectedKeys()
             .hasOnly(['paymentStatus','paidAmount','txnId','paymentNote','slip']);
    }
